cmake_minimum_required(VERSION 3.22)
project(SPEAKIT VERSION 0.4.2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "" FORCE)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
  add_subdirectory(third_party/JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(SPEAKIT
  COMPANY_NAME "Yishay Raziel" COMPANY_WEBSITE "https://yraziel.com" BUNDLE_ID com.yraziel.anallog
  PLUGIN_MANUFACTURER_CODE Yraz PLUGIN_CODE Anlg
  IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  FORMATS VST3 AU Standalone VST3_CATEGORIES Fx Distortion AU_MAIN_TYPE kAudioUnitType_MusicEffect
  PRODUCT_NAME "Anal-Log")
file(GLOB IRS CONFIGURE_DEPENDS Resources/ir/*.flac Resources/bay/*.ogg Resources/skin/*.png Resources/skin/*.jpg)
juce_add_binary_data(SPEAKITData SOURCES ${IRS} Resources/fonts/BarlowCondensed-600.ttf Resources/fonts/BarlowCondensed-700.ttf Resources/fonts/VT323-Regular.ttf THIRD_PARTY_NOTICES.txt)
set(TP third_party)
file(GLOB GSM_SRC ${TP}/libgsm/src/*.c)
list(FILTER GSM_SRC EXCLUDE REGEX "toast|debug|gsm_print|gsm_explode|gsm_implode")
set(SPX ${TP}/speex/libspeex)
set(SPEEX_SRC bits cb_search exc_10_16_table exc_10_32_table exc_20_32_table exc_5_256_table exc_5_64_table exc_8_128_table filters gain_table gain_table_lbr hexc_10_32_table hexc_table high_lsp_tables lpc lsp lsp_tables_nb ltp modes modes_wb nb_celp quant_lsp sb_celp speex speex_callbacks speex_header stereo vbr vq window)
list(TRANSFORM SPEEX_SRC PREPEND "${SPX}/")
list(TRANSFORM SPEEX_SRC APPEND ".c")
add_library(sk_codecs STATIC ${GSM_SRC} ${SPEEX_SRC})
target_include_directories(sk_codecs PUBLIC ${TP}/libgsm/inc ${TP}/speex/include Source/codec_cfg/speex ${SPX})
target_compile_definitions(sk_codecs PRIVATE FLOATING_POINT USE_ALLOCA HAVE_ALLOCA_H=1 EXPORT= SASR NeedFunctionPrototypes=1 NDEBUG)
set_target_properties(sk_codecs PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_definitions(sk_codecs PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
target_sources(SPEAKIT PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SPEAKIT PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_USE_FLAC=1 JUCE_USE_OGGVORBIS=1)
target_link_libraries(SPEAKIT PRIVATE SPEAKITData sk_codecs juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
option(SK_TESTS "Build tests" OFF)
if(SK_TESTS)
  target_compile_definitions(SPEAKIT PUBLIC JUCE_MODAL_LOOPS_PERMITTED=1)
  add_executable(sk_qa tests/qa.cpp)
  target_include_directories(sk_qa PRIVATE Source $<TARGET_PROPERTY:SPEAKIT,INCLUDE_DIRECTORIES>)
  target_compile_definitions(sk_qa PRIVATE $<TARGET_PROPERTY:SPEAKIT,COMPILE_DEFINITIONS> JucePlugin_Build_Standalone=0)
  target_link_libraries(sk_qa PRIVATE SPEAKIT)
  add_executable(sk_nan tests/nan.cpp)
  target_include_directories(sk_nan PRIVATE Source $<TARGET_PROPERTY:SPEAKIT,INCLUDE_DIRECTORIES>)
  target_compile_definitions(sk_nan PRIVATE $<TARGET_PROPERTY:SPEAKIT,COMPILE_DEFINITIONS> JucePlugin_Build_Standalone=0)
  target_link_libraries(sk_nan PRIVATE SPEAKIT)
  add_executable(sk_tests tests/test.cpp)
  target_include_directories(sk_tests PRIVATE Source $<TARGET_PROPERTY:SPEAKIT,INCLUDE_DIRECTORIES>)
  target_compile_definitions(sk_tests PRIVATE $<TARGET_PROPERTY:SPEAKIT,COMPILE_DEFINITIONS> JucePlugin_Build_Standalone=0)
  target_link_libraries(sk_tests PRIVATE SPEAKIT)
endif()
